  <script>
    (function() {
      var SUPABASE_URL = '%%SUPABASE_URL%%';
      var SUPABASE_ANON_KEY = '%%SUPABASE_ANON_KEY%%';

      var supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

      var loadingEl = document.getElementById('loading');
      var formEl = document.getElementById('password-form');
      var errorStateEl = document.getElementById('error-state');
      var errorMsgEl = document.getElementById('error-msg');
      var successStateEl = document.getElementById('success-state');
      var welcomeMsgEl = document.getElementById('welcome-msg');
      var passwordInput = document.getElementById('password');
      var confirmInput = document.getElementById('confirm-password');
      var submitBtn = document.getElementById('submit-btn');
      var statusEl = document.getElementById('status');
      var strengthFill = document.getElementById('strength-fill');

      function show(el) { el.classList.remove('hidden'); }
      function hide(el) { el.classList.add('hidden'); }

      function showError(msg) {
        hide(loadingEl); hide(formEl); hide(successStateEl);
        errorMsgEl.textContent = msg;
        show(errorStateEl);
      }

      function showForm(userEmail) {
        hide(loadingEl); hide(errorStateEl); hide(successStateEl);
        welcomeMsgEl.innerHTML = 'Enter a new password for <strong>' + (userEmail || 'your account') + '</strong>.';
        show(formEl);
        passwordInput.focus();
      }

      function showSuccess() {
        hide(loadingEl); hide(formEl); hide(errorStateEl);
        show(successStateEl);
        setTimeout(function() { window.location.href = '/'; }, 2500);
      }

      // Password strength indicator
      passwordInput.addEventListener('input', function() {
        var val = this.value;
        var score = 0;
        if (val.length >= 8) score++;
        if (val.length >= 12) score++;
        if (/[A-Z]/.test(val) && /[a-z]/.test(val)) score++;
        if (/\d/.test(val)) score++;
        if (/[^A-Za-z0-9]/.test(val)) score++;

        var pct = Math.min(score / 5 * 100, 100);
        var color = pct <= 20 ? '#ef4444' : pct <= 40 ? '#f97316' : pct <= 60 ? '#eab308' : pct <= 80 ? '#22c55e' : '#16a34a';
        strengthFill.style.width = pct + '%';
        strengthFill.style.background = color;
      });

      // Submit handler
      formEl.addEventListener('submit', function(e) {
        e.preventDefault();
      });

      submitBtn.addEventListener('click', async function() {
        var pw = passwordInput.value;
        var confirm = confirmInput.value;

        // Reset any prior status. Use '' (not 'none') so the CSS
        // display:block from .status.error/.success can win. Inline
        // display:none was silently hiding every subsequent message.
        statusEl.className = 'status';
        statusEl.style.display = '';
        statusEl.textContent = '';

        if (!pw || pw.length < 8) {
          statusEl.className = 'status error';
          statusEl.textContent = 'Password must be at least 8 characters.';
          return;
        }
        if (pw !== confirm) {
          statusEl.className = 'status error';
          statusEl.textContent = 'Passwords do not match.';
          return;
        }

        submitBtn.disabled = true;
        submitBtn.textContent = 'Resetting...';

        var result = await supabase.auth.updateUser({ password: pw });
        if (result.error) {
          statusEl.className = 'status error';
          statusEl.textContent = 'Error: ' + result.error.message;
          submitBtn.disabled = false;
          submitBtn.textContent = 'Reset Password';
          return;
        }

        showSuccess();
      });

      // ── Initialize: check for recovery tokens in URL hash ──
      async function init() {
        var hash = window.location.hash;

        if (hash && hash.includes('access_token')) {
          // Supabase puts recovery tokens in the URL hash:
          // #access_token=...&refresh_token=...&type=recovery
          var result = await supabase.auth.getSession();
          if (result.data && result.data.session) {
            showForm(result.data.session.user.email || '');
            return;
          }
          // Try setting session manually from hash params
          var params = new URLSearchParams(hash.substring(1));
          var accessToken = params.get('access_token');
          var refreshToken = params.get('refresh_token');
          if (accessToken && refreshToken) {
            var setResult = await supabase.auth.setSession({
              access_token: accessToken,
              refresh_token: refreshToken,
            });
            if (setResult.data && setResult.data.session) {
              showForm(setResult.data.session.user.email || '');
              return;
            }
          }
          showError('Your reset link appears to be invalid or expired. Please request a new one.');
          return;
        }

        // No hash tokens — check if already logged in
        var session = await supabase.auth.getSession();
        if (session.data && session.data.session) {
          showForm(session.data.session.user.email || '');
          return;
        }

        showError('No reset token found. Please use the link from your password reset email, or request a new one.');
      }

      init();
    })();
  </script>
