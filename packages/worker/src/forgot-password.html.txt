  <script>
    var supabaseUrl = '%%SUPABASE_URL%%';
    var supabaseKey = '%%SUPABASE_ANON_KEY%%';
    var sb = window.supabase.createClient(supabaseUrl, supabaseKey);

    // Track the email across steps so we can call verifyOtp with it.
    var currentEmail = '';

    // Show status message. `kind` is 'success' | 'error' | 'info'.
    function showStatus(text, kind) {
      var s = document.getElementById('status');
      s.className = 'status ' + (kind || 'info');
      s.innerHTML = text;
      s.classList.remove('hidden');
    }
    function clearStatus() {
      var s = document.getElementById('status');
      s.classList.add('hidden');
      s.innerHTML = '';
    }

    function showStep(id) {
      ['step-email', 'step-code', 'step-done'].forEach(function(x) {
        document.getElementById(x).classList.toggle('hidden', x !== id);
      });
    }

    // ─── Step 1: request the code ────────────────────────────
    async function handleSendCode() {
      var email = document.getElementById('email').value.trim();
      var btn = document.getElementById('send-btn');
      clearStatus();

      if (!email || !email.includes('@')) {
        showStatus('Please enter a valid email address.', 'error');
        return;
      }

      btn.disabled = true;
      btn.textContent = 'Sending...';

      try {
        // resetPasswordForEmail sends the Recovery email. Whether the user
        // gets a link, a code, or both depends on the template configured
        // in Supabase Studio. Our Recovery template includes {{ .Token }}.
        // No redirectTo needed since we're not using the link.
        var result = await sb.auth.resetPasswordForEmail(email);

        if (result.error) {
          // Don't leak whether the email exists — still show success UI.
          // Real errors (rate limit, invalid format) will bubble up here.
          if (/rate limit|too many/i.test(result.error.message)) {
            showStatus('Too many requests. Wait a minute and try again.', 'error');
            btn.disabled = false;
            btn.textContent = 'Send Code';
            return;
          }
          // Silent fail on other errors — proceed to code step anyway.
          console.warn('resetPasswordForEmail:', result.error.message);
        }

        currentEmail = email;
        document.getElementById('email-echo').textContent = 'Code sent to ' + email;
        showStep('step-code');
        showStatus('If an account exists for that email, a 6-digit code is on the way. Check your junk folder if you don\'t see it in 30 seconds.', 'success');
        setTimeout(function() {
          document.getElementById('code').focus();
        }, 100);
      } catch (err) {
        showStatus('Something went wrong. Please try again.', 'error');
        btn.disabled = false;
        btn.textContent = 'Send Code';
      }
    }

    // ─── Step 2: verify code + set new password ─────────────
    async function handleReset() {
      var code = document.getElementById('code').value.trim();
      var pw = document.getElementById('password').value;
      var confirm = document.getElementById('confirm-password').value;
      var btn = document.getElementById('reset-btn');
      clearStatus();

      if (!code || !/^\d{6}$/.test(code)) {
        showStatus('Enter the 6-digit code from your email.', 'error');
        return;
      }
      if (!pw || pw.length < 8) {
        showStatus('Password must be at least 8 characters.', 'error');
        return;
      }
      if (pw !== confirm) {
        showStatus('Passwords don\'t match.', 'error');
        return;
      }

      btn.disabled = true;
      btn.textContent = 'Verifying...';

      try {
        // Verify the OTP. Success returns a real session (user is logged in).
        var verify = await sb.auth.verifyOtp({
          email: currentEmail,
          token: code,
          type: 'recovery',
        });
        if (verify.error) {
          if (/expired/i.test(verify.error.message)) {
            showStatus('That code has expired. Click Resend to get a new one.', 'error');
          } else if (/invalid|not.found/i.test(verify.error.message)) {
            showStatus('That code isn\'t valid. Double-check the digits and try again.', 'error');
          } else {
            showStatus(verify.error.message, 'error');
          }
          btn.disabled = false;
          btn.textContent = 'Reset Password & Sign In';
          return;
        }

        // Now update the password. Also clear the must_reset_password flag
        // in case this was an admin-triggered force-reset — otherwise the
        // portal would immediately redirect them back here on next load.
        var update = await sb.auth.updateUser({
          password: pw,
          data: { must_reset_password: false },
        });
        if (update.error) {
          showStatus('Password couldn\'t be set: ' + update.error.message, 'error');
          btn.disabled = false;
          btn.textContent = 'Reset Password & Sign In';
          return;
        }

        showStep('step-done');
        clearStatus();
        // Session already active from verifyOtp; go home.
        setTimeout(function() {
          window.location.href = '/';
        }, 1500);
      } catch (err) {
        showStatus('Something went wrong. Please try again.', 'error');
        btn.disabled = false;
        btn.textContent = 'Reset Password & Sign In';
      }
    }

    // ─── Resend the code ────────────────────────────────────
    async function handleResend(e) {
      if (e) e.preventDefault();
      if (!currentEmail) return;
      var link = document.getElementById('resend-link');
      link.textContent = 'Sending...';
      link.style.pointerEvents = 'none';
      try {
        await sb.auth.resetPasswordForEmail(currentEmail);
        showStatus('New code sent to ' + currentEmail + '.', 'success');
      } catch (err) {
        showStatus('Couldn\'t resend. Try again in a minute.', 'error');
      } finally {
        setTimeout(function() {
          link.textContent = 'Resend';
          link.style.pointerEvents = '';
        }, 3000);
      }
    }

    // ─── Back button ────────────────────────────────────────
    function handleBackToEmail() {
      clearStatus();
      currentEmail = '';
      document.getElementById('email').value = '';
      document.getElementById('code').value = '';
      document.getElementById('password').value = '';
      document.getElementById('confirm-password').value = '';
      document.getElementById('send-btn').disabled = false;
      document.getElementById('send-btn').textContent = 'Send Code';
      showStep('step-email');
      setTimeout(function() {
        document.getElementById('email').focus();
      }, 100);
    }

    // ─── Enter-key handlers ─────────────────────────────────
    document.getElementById('email').addEventListener('keydown', function(e) {
      if (e.key === 'Enter') handleSendCode();
    });
    document.getElementById('confirm-password').addEventListener('keydown', function(e) {
      if (e.key === 'Enter') handleReset();
    });
    // Auto-strip non-digits from the OTP field as the user types.
    document.getElementById('code').addEventListener('input', function(e) {
      e.target.value = e.target.value.replace(/\D/g, '').slice(0, 6);
    });

    // Auto-focus email on load.
    document.getElementById('email').focus();
  </script>
