  <script>
    var supabaseUrl = '%%SUPABASE_URL%%';
    var supabaseKey = '%%SUPABASE_ANON_KEY%%';
    var sb = window.supabase.createClient(supabaseUrl, supabaseKey);

    function showStatus(text, kind) {
      var s = document.getElementById('status');
      s.className = 'status ' + (kind || 'info');
      s.textContent = text;
      s.classList.remove('hidden');
    }
    function clearStatus() {
      var s = document.getElementById('status');
      s.classList.add('hidden');
      s.textContent = '';
    }

    async function handleAccept() {
      var email = document.getElementById('email').value.trim();
      var code = document.getElementById('code').value.trim();
      var pw = document.getElementById('password').value;
      var confirm = document.getElementById('confirm-password').value;
      var btn = document.getElementById('submit-btn');
      clearStatus();

      if (!email || !email.includes('@')) {
        showStatus('Enter the email address the invite was sent to.', 'error');
        return;
      }
      if (!/^\d{6}$/.test(code)) {
        showStatus('Enter the 6-digit code from your invite email.', 'error');
        return;
      }
      if (!pw || pw.length < 8) {
        showStatus('Password must be at least 8 characters.', 'error');
        return;
      }
      if (pw !== confirm) {
        showStatus("Passwords don't match.", 'error');
        return;
      }

      btn.disabled = true;
      btn.textContent = 'Verifying...';

      try {
        // Invite type: user is being created (or accepted). verifyOtp
        // returns a real session. After that, we set their password.
        var verify = await sb.auth.verifyOtp({
          email: email,
          token: code,
          type: 'invite',
        });
        if (verify.error) {
          if (/expired/i.test(verify.error.message)) {
            showStatus('That code has expired. Ask your admin to send a new invite.', 'error');
          } else if (/invalid|not.found/i.test(verify.error.message)) {
            showStatus("That code isn't valid. Double-check the digits, and confirm the email matches the invite.", 'error');
          } else {
            showStatus(verify.error.message, 'error');
          }
          btn.disabled = false;
          btn.textContent = 'Accept Invite & Sign In';
          return;
        }

        // Set the password. Also clear must_reset_password in case the
        // admin flow set it — the OTP-based invite flow is a full
        // password set, not a "temp then reset" pattern.
        var update = await sb.auth.updateUser({
          password: pw,
          data: { must_reset_password: false },
        });
        if (update.error) {
          showStatus("Password couldn't be set: " + update.error.message, 'error');
          btn.disabled = false;
          btn.textContent = 'Accept Invite & Sign In';
          return;
        }

        document.getElementById('form-section').classList.add('hidden');
        document.getElementById('success-section').classList.remove('hidden');
        clearStatus();
        setTimeout(function() { window.location.href = '/'; }, 1500);
      } catch (err) {
        showStatus('Something went wrong. Please try again.', 'error');
        btn.disabled = false;
        btn.textContent = 'Accept Invite & Sign In';
      }
    }

    document.getElementById('confirm-password').addEventListener('keydown', function(e) {
      if (e.key === 'Enter') handleAccept();
    });
    document.getElementById('code').addEventListener('input', function(e) {
      e.target.value = e.target.value.replace(/\D/g, '').slice(0, 6);
    });

    // Pre-fill email from ?email=... if the invite email included a
    // convenience link. Otherwise leave empty for user to type.
    var params = new URLSearchParams(window.location.search);
    var preEmail = params.get('email');
    if (preEmail) document.getElementById('email').value = preEmail;

    document.getElementById(preEmail ? 'code' : 'email').focus();
  </script>
