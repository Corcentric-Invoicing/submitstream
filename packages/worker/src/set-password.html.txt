  <script>
    (function() {
      var SUPABASE_URL = '%%SUPABASE_URL%%';
      var SUPABASE_ANON_KEY = '%%SUPABASE_ANON_KEY%%';

      var supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

      var loadingEl = document.getElementById('loading');
      var formEl = document.getElementById('password-form');
      var errorStateEl = document.getElementById('error-state');
      var errorMsgEl = document.getElementById('error-msg');
      var successStateEl = document.getElementById('success-state');
      var welcomeMsgEl = document.getElementById('welcome-msg');
      var passwordInput = document.getElementById('password');
      var confirmInput = document.getElementById('confirm-password');
      var submitBtn = document.getElementById('submit-btn');
      var statusEl = document.getElementById('status');
      var strengthFill = document.getElementById('strength-fill');

      function show(el) { el.classList.remove('hidden'); }
      function hide(el) { el.classList.add('hidden'); }

      function showError(msg) {
        hide(loadingEl); hide(formEl); hide(successStateEl);
        errorMsgEl.textContent = msg;
        show(errorStateEl);
      }

      function showForm(userEmail) {
        hide(loadingEl); hide(errorStateEl); hide(successStateEl);
        welcomeMsgEl.innerHTML = 'Welcome! Set a password for <strong>' + (userEmail || 'your account') + '</strong> to access the SubmitStream portal.';
        show(formEl);
        passwordInput.focus();
      }

      function showSuccess() {
        hide(loadingEl); hide(formEl); hide(errorStateEl);
        show(successStateEl);
        setTimeout(function() { window.location.href = '/'; }, 2000);
      }

      // Password strength indicator
      passwordInput.addEventListener('input', function() {
        var val = this.value;
        var score = 0;
        if (val.length >= 8) score++;
        if (val.length >= 12) score++;
        if (/[A-Z]/.test(val) && /[a-z]/.test(val)) score++;
        if (/\d/.test(val)) score++;
        if (/[^A-Za-z0-9]/.test(val)) score++;

        var pct = Math.min(score / 5 * 100, 100);
        var color = pct <= 20 ? '#ef4444' : pct <= 40 ? '#f97316' : pct <= 60 ? '#eab308' : pct <= 80 ? '#22c55e' : '#16a34a';
        strengthFill.style.width = pct + '%';
        strengthFill.style.background = color;
      });

      // Submit handler
      formEl.addEventListener('submit', function(e) {
        e.preventDefault();
      });

      submitBtn.addEventListener('click', async function() {
        var pw = passwordInput.value;
        var confirm = confirmInput.value;

        // Reset any prior status. Use '' (not 'none') so the CSS
        // display:block from .status.error/.success can win. Inline
        // display:none was silently hiding every subsequent message.
        statusEl.className = 'status';
        statusEl.style.display = '';
        statusEl.textContent = '';

        if (!pw || pw.length < 8) {
          statusEl.className = 'status error';
          statusEl.textContent = 'Password must be at least 8 characters.';
          return;
        }
        if (pw !== confirm) {
          statusEl.className = 'status error';
          statusEl.textContent = 'Passwords do not match.';
          return;
        }

        submitBtn.disabled = true;
        submitBtn.textContent = 'Setting password...';

        // Update password AND clear must_reset_password flag in user_metadata.
        // The admin-set-temp-password flow flips that flag to true; we clear
        // it here so the portal stops redirecting back to /set-password on
        // every subsequent sign-in.
        var result = await supabase.auth.updateUser({
          password: pw,
          data: { must_reset_password: false }
        });
        if (result.error) {
          statusEl.className = 'status error';
          statusEl.textContent = 'Error: ' + result.error.message;
          submitBtn.disabled = false;
          submitBtn.textContent = 'Set Password & Sign In';
          return;
        }

        showSuccess();
      });

      // Sign-out escape hatch — see comment on the link element above.
      document.getElementById('sign-out-link').addEventListener('click', async function(e) {
        e.preventDefault();
        try { await supabase.auth.signOut(); } catch (_) { /* fall through */ }
        window.location.href = '/login';
      });

      // ── Initialize: check for invite/recovery tokens ──
      async function init() {
        // Supabase puts tokens (or errors) in the URL hash after redirect:
        //   Success: #access_token=...&refresh_token=...&type=invite
        //   Failure: #error=access_denied&error_code=otp_expired&error_description=...
        var hash = window.location.hash;

        // Detect Supabase-reported failure and show the real reason.
        // Otherwise the page would fall through to the generic "no token"
        // message and users think we lost the link, not that it expired.
        if (hash && hash.includes('error=')) {
          var errParams = new URLSearchParams(hash.substring(1));
          var errCode = errParams.get('error_code') || errParams.get('error') || '';
          var errDesc = (errParams.get('error_description') || '').replace(/\+/g, ' ');
          if (errCode === 'otp_expired' || /expired/i.test(errDesc)) {
            showError('This invite/reset link has expired. Ask your admin to send a fresh one — links are single-use and time out after a few hours.');
          } else if (errCode === 'access_denied') {
            showError('This link has already been used or is invalid. If you already set a password, sign in from the portal. Otherwise ask your admin to send a new invite.');
          } else {
            showError('The link is invalid: ' + (errDesc || errCode || 'unknown error') + '. Ask your admin to send a new one.');
          }
          return;
        }

        if (hash && hash.includes('access_token')) {
          // Let Supabase client pick up the tokens from the hash
          var result = await supabase.auth.getSession();
          if (result.data && result.data.session) {
            var email = result.data.session.user.email || '';
            showForm(email);
            return;
          }
          // If getSession didn't work, try setSession manually
          var params = new URLSearchParams(hash.substring(1));
          var accessToken = params.get('access_token');
          var refreshToken = params.get('refresh_token');
          if (accessToken && refreshToken) {
            var setResult = await supabase.auth.setSession({
              access_token: accessToken,
              refresh_token: refreshToken,
            });
            if (setResult.data && setResult.data.session) {
              showForm(setResult.data.session.user.email || '');
              return;
            }
          }
          showError('Your invite link appears to be invalid or expired. Please ask your admin to send a new invite.');
          return;
        }

        // No hash tokens — check if user is already logged in (maybe navigated here directly)
        var session = await supabase.auth.getSession();
        if (session.data && session.data.session) {
          showForm(session.data.session.user.email || '');
          return;
        }

        showError('No invite token found. Please use the link from your invite email, or ask your admin to resend it.');
      }

      init();
    })();
  </script>
